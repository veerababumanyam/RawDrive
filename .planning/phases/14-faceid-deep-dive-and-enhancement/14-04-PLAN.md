         interface NormalizedResponse<T> {
           data: T[];
           meta: { total: number; page: number; per_page: number; };
         }

         function normalizePaginatedResponse<T>(response: unknown): NormalizedResponse<T> {
           if (Array.isArray(response)) {
             return { data: response as T[], meta: { total: response.length, page: 1, per_page: response.length } };
           }
           if (response && typeof response === 'object' && 'data' in response) {
             const r = response as { data: T[]; meta?: any };
             return { data: r.data, meta: r.meta ?? { total: r.data.length, page: 1, per_page: r.data.length } };
           }
           return { data: [], meta: { total: 0, page: 1, per_page: 0 } };
         }
