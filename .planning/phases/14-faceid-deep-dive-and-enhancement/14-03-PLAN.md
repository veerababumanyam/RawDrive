         async def increment_cache_version(workspace_id: UUID) -> int:
             return await redis.incr(CACHE_VERSION_KEY.format(workspace_id=workspace_id))

         async def get_cache_version(workspace_id: UUID) -> int:
             val = await redis.get(CACHE_VERSION_KEY.format(workspace_id=workspace_id))

             if l1_data and l1_version == current_version:
