Download API endpoints:
```
POST /public/galleries/{token}/downloads/single
  Body: { asset_id: UUID, size: "web"|"print"|"original", format: "jpeg"|"png"|"webp" }
  Response: { download_url: string }

POST /public/galleries/{token}/downloads/batch
  Body: { asset_ids: UUID[], size: string, include_favorites: bool, include_selections: bool }
  Response: { job_id: string }

GET /public/galleries/{token}/downloads/job/{job_id}
  Response: { job_id, status: "pending"|"processing"|"complete"|"failed", progress: 0-100, download_url?: string, expires_in?: number }

GET /public/galleries/{token}/downloads/job/{job_id}/stream
  SSE events: data: { status, progress, download_url? }

GET /galleries/{gallery_id}/downloads/log?page=1&per_page=50
  Response: { downloads: [{ id, asset_id, visitor_token, size, file_size_bytes, created_at }], total_count, total_bytes }
```
