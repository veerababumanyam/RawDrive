# =============================================================================
# RawDrive — HAProxy leader-routing config (deploy/haproxy/haproxy.cfg)
# =============================================================================
#
# ⚠️  PLANNED CUTOVER — INERT UNTIL DELIBERATELY ACTIVATED  ⚠️
#
# HAProxy is the transparent leader-router for the Patroni automatic-failover
# subsystem. It REPLACES the manual "edit databases.ini + reload pgbouncer" flip
# (deploy/scripts/promote-postgres-replica.sh + docs/runbooks/postgres-failover.md).
# It is started ONLY during the maintenance-window cutover in
# docs/runbooks/patroni-failover.md. Until cutover, pgbouncer points directly at
# the fixed primary IP and the manual model is in force.
#
# -----------------------------------------------------------------------------
# What this does, and why pgbouncer no longer needs hand-editing on failover
# -----------------------------------------------------------------------------
# Each Patroni member exposes a REST API on :8008 that answers:
#     GET /primary  → HTTP 200 ONLY on the node that is currently the leader,
#                     503 on every standby.
#     GET /replica  → HTTP 200 ONLY on a running standby (read-only), 503 else.
#
# HAProxy health-checks BOTH Patroni members on /primary. Because exactly ONE
# member returns 200 at any instant (etcd guarantees a single leader), HAProxy's
# write frontend (:5432) always forwards to whichever node is the current
# leader. When Patroni fails over, the OLD leader's /primary starts returning 503
# and the NEW leader's starts returning 200 — HAProxy reroutes within a couple of
# health-check intervals, with NO human and NO config edit.
#
# So pgbouncer's databases.ini is changed ONCE, at cutover, to point at the LOCAL
# HAProxy (host=127.0.0.1 port=5432) instead of the fixed DB-node IP. After that
# it NEVER needs editing on failover again — HAProxy absorbs the leader change
# behind a stable local address. (See the integration notes for the exact
# databases.ini line.)
#
# -----------------------------------------------------------------------------
# Where this runs
# -----------------------------------------------------------------------------
# HAProxy runs on the APP nodes (.42 and .44) — co-located with pgbouncer — and
# publishes :5432 / :5433 / :7000 on LOOPBACK only (127.0.0.1), because its only
# client is the local pgbouncer on the same host. It health-checks the Patroni
# REST APIs on .46 and .44 cross-host. Running an HAProxy on EACH app node (not a
# single shared one) avoids making HAProxy itself a single point of failure: if
# one app node dies, the other app node's pgbouncer→HAProxy path is unaffected.
#
# Pinned image: haproxy:3.0.6-alpine (see compose). Never :latest.
# =============================================================================

global
    # Run in the foreground (Docker owns process supervision; no daemon fork).
    log stdout format raw local0 info
    maxconn 2000
    # Drop privileges after binding — defense in depth even inside a container.
    user  haproxy
    group haproxy
    # Tune the runtime stats socket for `patroni-status.sh` / live inspection.
    stats socket /var/run/haproxy.sock mode 660 level admin
    stats timeout 30s

defaults
    log     global
    mode    tcp                 # Postgres is a raw TCP protocol; HAProxy does NOT
                                # speak SQL — it proxies bytes and routes purely on
                                # the Patroni REST health check below.
    retries 2
    # Timeouts: generous server timeout because long analytical/admin queries can
    # legitimately run. The connect timeout is short so a dead leader is shed
    # fast. The check timeout bounds the /primary probe.
    timeout connect 5s
    timeout client  30m
    timeout server  30m
    timeout check   5s
    # http-keep-alive only matters for the httpchk health probe, which is HTTP.
    default-server inter 3s fall 3 rise 2 on-marked-down shutdown-sessions
    # ^ inter 3s : probe every 3s.  fall 3 : 3 failed probes mark DOWN.
    #   rise 2  : 2 good probes mark UP.  on-marked-down shutdown-sessions:
    #   when the OLD leader is marked down (post-failover), KILL its existing TCP
    #   sessions immediately so pgbouncer reconnects to the NEW leader instead of
    #   hanging on a dead/demoted backend. This is what makes failover fast for
    #   already-open pooled connections.

# =============================================================================
# WRITE traffic — :5432 → current PRIMARY (leader)
# =============================================================================
# pgbouncer's databases.ini points here (127.0.0.1:5432). HAProxy probes both
# Patroni members' REST APIs with `GET /primary`; only the leader answers 200, so
# only the leader is "UP" in this backend and receives writes. Exactly one node
# is ever UP → no split-brain at the routing layer.
frontend pg_primary_frontend
    bind 127.0.0.1:5432
    bind ${HOST_GATEWAY_IP}:5432
    bind ${APP_BRIDGE_GATEWAY_IP}:5432
    mode tcp
    default_backend pg_primary_backend

backend pg_primary_backend
    mode tcp
    # Probe the Patroni REST API (HTTP) even though the data path is TCP.
    option httpchk
    http-check send meth GET uri /primary
    http-check expect status 200
    # Both DB-capable members. `port 8008` aims the HTTP health check at the
    # Patroni REST API while the actual proxied traffic goes to `:5432` on the
    # SAME host (the address before the colon). Only the leader's /primary
    # returns 200, so only the leader is UP here.
    # NOTE: ${...} placeholders are env-substituted at container start by the
    # compose command (envsubst) so the IPs come from /opt/rawdrive/app/.env,
    # never hardcoded. The member set is the two DB-capable nodes: .46 and .44.
    server pg-46 ${ETCD_NODE_46_IP}:5432 check port 8008
    server pg-44 ${ETCD_NODE_44_IP}:5432 check port 8008

# =============================================================================
# READ traffic — :5433 → any healthy STANDBY (replica)
# =============================================================================
# OPTIONAL read-scaling path. Probe `GET /replica`; standbys answer 200. Useful
# for read-only reporting/analytics that can tolerate a few ms of replication
# lag. The app does NOT have to use this — it is offered so read load can be
# pointed at :5433 deliberately (e.g. a read-only pgbouncer database entry).
# leastconn balances across multiple standbys if the cluster ever grows past 2.
frontend pg_replica_frontend
    bind 127.0.0.1:5433
    bind ${HOST_GATEWAY_IP}:5433
    bind ${APP_BRIDGE_GATEWAY_IP}:5433
    mode tcp
    default_backend pg_replica_backend

backend pg_replica_backend
    mode tcp
    balance leastconn
    option httpchk
    http-check send meth GET uri /replica
    http-check expect status 200
    # Same two members. Whichever is currently a standby answers /replica with
    # 200; the leader answers 503 and is excluded from the read pool. If there is
    # NO standby (degraded cluster), this backend has zero UP servers and reads
    # to :5433 fail fast — by design, callers must fall back to :5432.
    server pg-46 ${ETCD_NODE_46_IP}:5432 check port 8008
    server pg-44 ${ETCD_NODE_44_IP}:5432 check port 8008

# =============================================================================
# STATS — :7000 (loopback) read-only dashboard for patroni-status.sh / humans
# =============================================================================
frontend stats_frontend
    bind 127.0.0.1:7000
    mode http
    stats enable
    stats uri /
    stats refresh 5s
    stats show-legends
    # Read-only: no `stats admin` — operators must not toggle backends by hand
    # (that is Patroni's job). Loopback-only bind keeps it off the network.
